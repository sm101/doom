* Company, combine English and Serbian Latin
Uses aspell (install via =brew install aspell=) to dump word lists for =en= and =sr-latn=.
No dictionary files to download manually; aspell resolves its own data directory.
The combined list is generated once and cached in =doom-data-dir=.

#+begin_src elisp
(defun my/ensure-combined-dictionary ()
  "Generate combined EN+SR-Latin word list using aspell.
Dumps dictionaries via aspell at first use; result cached in `doom-data-dir'."
  (let* ((dict-dir (expand-file-name "dicts" doom-data-dir))
         (combined (expand-file-name "aspell-combined.txt" dict-dir))
         (aspell (executable-find "aspell")))
    (unless (file-exists-p combined)
      (unless aspell
        (user-error "aspell not found; install it with: brew install aspell"))
      (make-directory dict-dir t)
      (message "Generating combined dictionary via aspell (en + sr-latn)...")
      (with-temp-buffer
        (dolist (lang '("en" "sr-latn"))
          (call-process aspell nil '(t nil) nil "-d" lang "dump" "master"))
        (call-process-region (point-min) (point-max) "sort" t t nil "-u")
        (write-region (point-min) (point-max) combined))
      (message "Combined dictionary written to %s" combined))
    combined))

(after! company
  (setq company-ispell-dictionary (my/ensure-combined-dictionary)))
#+end_src
